Project WelcomeTPU {
  database_type: 'PostgreSQL'
}

Enum card_type {
  FACT
  EVENT
  STATISTIC
  PERSON
  MEDIA
  QR
}

Enum content_format {
  PLAIN
  MARKDOWN
  HTML
}

Table schools {
  id bigint [pk, increment]
  name varchar(255) [not null, unique]
  short_description text
  full_description text [not null]
  official_site_url varchar(1024)
  icon_asset_id varchar(64) [ref: > assets.id]
  created_at timestamp [not null, default: `now()`]
  updated_at timestamp [not null, default: `now()`]
}

Table cards {
  id uuid [pk, default: `gen_random_uuid()`]
  type card_type [not null]
  school_id bigint [ref: >? schools.id, note: 'null - карточка о тпу в целом']
  title varchar(1024) [not null]
  content text [not null]
  content_format content_format [not null, default: 'PLAIN']
  published_at timestamp [note: 'null - черновик']
  unique_hash varchar(64) [not null, unique, note: 'sha256 от type, school_id, title, content']
  created_by_user_id bigint [note: 'user_id из oauth тпу']
  created_at timestamp [not null, default: `now()`]
  updated_at timestamp [not null, default: `now()`]
}

Table card_sources {
  id bigint [pk, increment]
  card_id uuid [ref: > cards.id, not null]
  title varchar(255)
  url varchar(1024) [not null]
}

Table event_details {
  card_id uuid [pk, ref: - cards.id]
  start_date date [not null]
  end_date date
  start_time time
  end_time time
  location varchar(1024)
  details_url varchar(1024)
}

Table statistic_details {
  card_id uuid [pk, ref: - cards.id]
  value numeric(20, 4) [not null]
  unit varchar(50) [not null]
  period varchar(100)
  source varchar(500)
}

Table qr_details {
  card_id uuid [pk, ref: - cards.id]
  target_url varchar(1024) [not null, note: 'URL, зашитый в QR-код']
  caption varchar(255)
  source varchar(500)
}

Enum media_type {
  PHOTO
  VIDEO
}

Table person_details {
  card_id uuid [pk, ref: - cards.id]
  full_name varchar(255) [not null]
  photo_asset_id varchar(64) [ref: > assets.id]
  bio text [note: 'краткая биография']
  achievements text [note: 'достижения']
}

Table media_details {
  card_id uuid [pk, ref: - cards.id]
  media_type media_type [not null]
  asset_id varchar(64) [ref: > assets.id, note: 'фото или превью видео']
  external_url varchar(1024) [note: 'внешнее видео']
  caption varchar(255)
  description text
  source varchar(500)
}

Table assets {
  id varchar(64) [pk, note: 'sha256 содержимого файла']
  url varchar(1024) [not null]
  mime_type varchar(50) [not null, note: 'image/jpeg, image/png, image/webp']
  size_bytes bigint [not null]
  width int [not null]
  height int [not null]
  alt_text varchar(512)
  created_at timestamp [not null, default: `now()`]
}

Table card_assets {
  card_id uuid [ref: > cards.id]
  asset_id varchar(64) [ref: > assets.id]

  Indexes {
    (card_id, asset_id) [pk]
  }
}
